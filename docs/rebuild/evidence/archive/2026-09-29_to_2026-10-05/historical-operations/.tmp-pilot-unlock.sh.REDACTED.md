#!/usr/bin/env bash
set -euo pipefail
API='https://mayaos.ru/api'
SLUG='pilot-barber-2608081910'
EMAIL='[REDACTED IDENTITY]'
PASS=REDACTED

login() {
  curl -sS -X POST "$API/auth/login" -H 'Content-Type: application/json' \
    -d "{\"tenantSlug\":\"$SLUG\",\"email\":\"$EMAIL\",\"password\":\"$PASS\"}"
}

echo '== owner login =='
LOGIN=$(login)
TOKEN=$(printf '%s' "$LOGIN" | python3 -c 'import sys,json; print(json.load(sys.stdin).get("access_token") or "")')
echo "token_ok=$([ -n "$TOKEN" ] && echo yes || echo no)"
test -n "$TOKEN"
AUTH=( -H "Authorization: Bearer $TOKEN" )

echo '== SQL trialFullAccess =='
JUMP='ssh -i ~/.ssh/beget_deploy -W %h:%p [REDACTED IDENTITY]'
ssh -o ProxyCommand="$JUMP" -i ~/.ssh/yandex_bot -o BatchMode=yes [REDACTED IDENTITY] 'bash -s' <<'EOS'
set -euo pipefail
sudo -u maya-saas bash -lc 'set -a; source /etc/maya-saas/live-widgets.env; set +a; node <<'"'"'NODE'"'"'
const { Client } = require("pg");
(async () => {
  const c = new Client({ connectionString: process.env.DATABASE_URL });
  await c.connect();
  const u = await c.query(
    `UPDATE "Tenant" SET "trialFullAccess"=true, "allowSelfRegistration"=true
     WHERE slug=$1
     RETURNING slug, status, "trialFullAccess", "allowSelfRegistration", "calendarSource"`,
    ["pilot-barber-2608081910"],
  );
  console.log(JSON.stringify(u.rows[0]));
  await c.end();
})().catch((e) => { console.error(e); process.exit(1); });
NODE'
EOS

echo '== setup before =='
SETUP=$(curl -sS "$API/internal-calendar/setup" "${AUTH[@]}")
printf '%s' "$SETUP" | python3 -c 'import sys,json; d=json.load(sys.stdin); print("ready", d.get("ready"), "providers", len(d.get("providers") or []), "services", len(d.get("services") or [])); open("/tmp/pilot-setup.json","w").write(sys.argv[1] if False else "");'
printf '%s' "$SETUP" > /tmp/pilot-setup.json
python3 - <<'PY'
import json
d=json.load(open('/tmp/pilot-setup.json'))
print('keys', sorted(d.keys())[:20])
print('ready', d.get('ready'))
print('providers', [(p.get('id'), p.get('displayName') or p.get('display_name')) for p in (d.get('providers') or [])])
print('services', [(s.get('id'), s.get('name')) for s in (d.get('services') or [])])
PY

PID=$(python3 -c 'import json; d=json.load(open("/tmp/pilot-setup.json")); ps=d.get("providers") or []; print(ps[0]["id"] if ps else "")')
if [ -z "$PID" ]; then
  echo '== create provider =='
  PROV=$(curl -sS -X POST "$API/internal-calendar/providers" "${AUTH[@]}" -H 'Content-Type: application/json' \
    -d '{"displayName":"Мастер Пилот","title":"Барбер"}')
  printf '%s\n' "$PROV" | head -c 500; echo
  PID=$(printf '%s' "$PROV" | python3 -c 'import sys,json; print(json.load(sys.stdin).get("id") or "")')
fi
echo "PID=$PID"
test -n "$PID"

SID=$(python3 -c 'import json; d=json.load(open("/tmp/pilot-setup.json")); ss=d.get("services") or []; print(ss[0]["id"] if ss else "")')
if [ -z "$SID" ]; then
  echo '== create service =='
  SVC=$(curl -sS -X POST "$API/internal-calendar/services" "${AUTH[@]}" -H 'Content-Type: application/json' \
    -d '{"name":"Стрижка","price":1500,"durationMinutes":45}')
  printf '%s\n' "$SVC" | head -c 500; echo
  SID=$(printf '%s' "$SVC" | python3 -c 'import sys,json; print(json.load(sys.stdin).get("id") or "")')
fi
echo "SID=$SID"
test -n "$SID"

echo '== attach service to provider (patch provider) =='
curl -sS -X PATCH "$API/internal-calendar/providers/$PID" "${AUTH[@]}" -H 'Content-Type: application/json' \
  -d "{\"serviceIds\":[\"$SID\"]}" | head -c 400; echo || true

echo '== schedule =='
curl -sS -X PUT "$API/internal-calendar/providers/$PID/schedule" "${AUTH[@]}" -H 'Content-Type: application/json' \
  -d '{"rules":[{"weekday":0,"startTime":"10:00","endTime":"20:00"},{"weekday":1,"startTime":"10:00","endTime":"20:00"},{"weekday":2,"startTime":"10:00","endTime":"20:00"},{"weekday":3,"startTime":"10:00","endTime":"20:00"},{"weekday":4,"startTime":"10:00","endTime":"20:00"},{"weekday":5,"startTime":"10:00","endTime":"20:00"},{"weekday":6,"startTime":"10:00","endTime":"20:00"}]}' \
  | head -c 400; echo

echo '== setup after =='
curl -sS "$API/internal-calendar/setup" "${AUTH[@]}" | python3 -c 'import sys,json; d=json.load(sys.stdin); print("ready", d.get("ready"), "providers", len(d.get("providers") or []), "services", len(d.get("services") or []))'

echo '== mobile config =='
curl -sS "$API/mobile/config/$SLUG" | python3 -c 'import sys,json; d=json.load(sys.stdin); print({k:d.get(k) for k in ["client_registration_enabled","booking_live_enabled","booking_mode","guest_access_ready","guest_access_blockers"]})'
