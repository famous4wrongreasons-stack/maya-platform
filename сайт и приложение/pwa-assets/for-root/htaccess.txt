#  Мужская Эстетика — правила для скачивания APK
#  Место: КОРЕНЬ сайта malesthetic.pro/ (рядом с index.html и malesthetic.apk)

# Android распознаёт .apk как устанавливаемый пакет (а не «непонятный файл»)
<IfModule mod_mime.c>
  AddType application/vnd.android.package-archive .apk
</IfModule>

# Принудительно скачивать .apk + не кэшировать (чтобы новая сборка подхватывалась сразу)
<IfModule mod_headers.c>
  <FilesMatch "\.apk$">
    Header set Content-Disposition "attachment; filename=malesthetic.apk"
    Header set Cache-Control "no-cache, no-store, must-revalidate"
  </FilesMatch>
</IfModule>
